## 10-1. 멀티 에이전트 아키텍처 개요

```
Orchestrator Agent (Sonnet) — as_tool() 자율 라우팅
│
├─ ParsingAgent (Haiku, as_tool)     ← DOCX 파서 결과 → Contract JSON 정제
├─ RiskAgent (Sonnet, as_tool)       ← 리스크 탐지 + Diff + 재무 분석
├─ SearchAgent (Haiku, as_tool)      ← Bedrock KB 시맨틱 검색
└─ MCP Client (mcp-server-sqlite)    ← 사내 컴플라이언스 규정 자율 쿼리
```

**설계 원칙**: "AI가 필요한 곳에만 AI를 쓴다." — 확정적 로직(difflib Diff, 라우팅 규칙)은 코드로 처리하고, LLM은 비정형 텍스트 이해·리스크 판단·자연어 검색에만 투입.

import difflib
import json
import boto3

dynamodb = boto3.resource('dynamodb', region_name='ap-northeast-2')

        contract_type: 계약 유형 (SI, NDA, MSA 등)
    Returns:
        리스크 분석 결과 (risk_level, risk_type, reason, recommendation, financial_impact)
    """
    pass  # LLM이 실제 분석 수행

@tool
def diff_clauses(contract_id: str, current_version: int) -> dict:
    """이전 버전과 현재 버전의 조항을 비교하여 변경점을 추출한다.
    DynamoDB에서 이전 버전 조회 → difflib SequenceMatcher 조항 단위 비교 → 리스크 영향 요약.

    Args:
        contract_id: 계약서 고유 ID
        current_version: 현재 버전 번호
    Returns:
        Diff Report (변경된 조항 목록 + 리스크 영향 요약)
    """
    # DynamoDB에서 이전 버전 조회
    prev_resp = contracts_table.get_item(Key={"id": f"{contract_id}-v{current_version - 1}"})
    curr_resp = contracts_table.get_item(Key={"id": f"{contract_id}-v{current_version}"})

    prev_item = prev_resp.get("Item")
    curr_item = curr_resp.get("Item")
    if not prev_item or not curr_item:
        return {"error": "이전 버전을 찾을 수 없습니다"}

    old_clauses = json.loads(prev_item.get("clauses_json", "[]"))
    new_clauses = json.loads(curr_item.get("clauses_json", "[]"))

    # difflib SequenceMatcher로 조항 단위 비교
    diffs = []
    for old_c, new_c in zip(old_clauses, new_clauses):
        if old_c["content"] != new_c["content"]:
            matcher = difflib.SequenceMatcher(None, old_c["content"], new_c["content"])
            diffs.append({
                "clause_id": new_c["id"],
                "title": new_c["title"],
                "ratio": str(matcher.ratio()),
                "old_content": old_c["content"],
                "new_content": new_c["content"],
            })

    return {"diffs": diffs, "total_changes": len(diffs)}

@tool
def analyze_financials(financials: dict, contract_type: str) -> dict:

### SearchAgent Tools

```python
# tools/search_tools.py
from strands import tool
import boto3
import os

@tool
def search_history(query: str) -> dict:
    """Bedrock Knowledge Base에서 계약서 히스토리를 시맨틱 검색한다.

    Args:
        query: 자연어 검색 질의 (예: "과거 CR 조항 이력?")
    Returns:
        검색 결과 + 생성된 답변
    """
    client = boto3.client('bedrock-agent-runtime', region_name='us-west-2')
    response = client.retrieve_and_generate(
        input={"text": query},
        retrieveAndGenerateConfiguration={
            "type": "KNOWLEDGE_BASE",
            "knowledgeBaseConfiguration": {
                "knowledgeBaseId": os.getenv("BEDROCK_KB_ID"),
                "modelArn": f"arn:aws:bedrock:us-west-2::foundation-model/anthropic.claude-sonnet-4-20250514"
            }
        }
    )
    return {
        "answer": response["output"]["text"],
        "citations": [c["retrievedReferences"] for c in response.get("citations", [])]
    }
```

## 10-5. DOCX 파서 (코드 기반, LLM 불필요)

```python
# parsers/docx_parser.py
import re
from docx import Document

def parse_docx(file_path: str) -> dict:
    """DOCX → Contract JSON (python-docx + regex, LLM 없이)"""
    doc = Document(file_path)
    full_text = "\n".join([p.text for p in doc.paragraphs])

    return {
        "contract_type": detect_type(full_text),
        "is_standard": detect_standard(full_text),
        "parties": extract_parties(full_text),
        "dates": extract_dates(full_text),
        "financials": extract_financials(full_text),
        "clauses": extract_clauses(doc)
    }

def detect_type(text: str) -> str:
    patterns = {
        "NDA": r"비밀유지|기밀유지|NDA",
        "MSA": r"기본계약|MSA|Master Service",
        "SI": r"도급|SI|시스템 ?통합|소프트웨어 ?개발",
        "SLA": r"SLA|서비스 ?수준",
        "Maintenance": r"유지보수|유지관리",
    }
    for ctype, pattern in patterns.items():
        if re.search(pattern, text, re.IGNORECASE):
            return ctype
    return "Other"

def extract_clauses(doc) -> list:
    """제N조 패턴으로 조항 분리"""
    clauses, current = [], None
    clause_pattern = re.compile(r"^제\s*(\d+)\s*조")

    for para in doc.paragraphs:
        match = clause_pattern.match(para.text.strip())
        if match:
            if current:
                clauses.append(current)
            current = {
                "id": f"clause_{match.group(1).zfill(3)}",
                "title": para.text.strip(),
                "content": "",
                "paragraph": int(match.group(1))
            }
        elif current:
            current["content"] += para.text + "\n"

    if current:
        clauses.append(current)
    return clauses
```

## 10-6. Rule Engine (코드 기반, LLM 불필요)

```python
# engine/rule_engine.py

def route_workflow(risk_report: dict, contract: dict) -> list:
    """MZC 내부 검토 라우팅 규칙 (if/else, LLM 불필요)"""
    steps = []
    overall = risk_report.get("overall_risk", "LOW")

    if overall == "HIGH":
        steps = [
            {"department": "계약팀", "step_order": 1},
            {"department": "법무팀", "step_order": 2},
            {"department": "본부장", "step_order": 3},
        ]
    elif overall == "MEDIUM":
        steps = [
            {"department": "계약팀", "step_order": 1},
            {"department": "담당 임원", "step_order": 2},
        ]
    else:
        steps = [{"department": "영업팀 담당자", "step_order": 1}]

    # 계약금액 10억+ → 재무팀 병렬 검토
    amount = contract.get("financials", {}).get("total_amount", 0)
    if amount >= 1_000_000_000:
        steps.append({"department": "재무팀", "step_order": 1, "parallel": True})

    # IP 완전이전 → 기술법무 순차 추가
    for cr in risk_report.get("clause_risks", []):
        if cr.get("risk_type") == "IP_완전이전":
            max_order = max(s["step_order"] for s in steps)
            steps.append({"department": "기술법무", "step_order": max_order + 1})
            break

    # 비표준계약 → 계약팀 에스컬레이션
    if not contract.get("is_standard", True):
        steps[0]["escalation"] = True

    return steps
```
